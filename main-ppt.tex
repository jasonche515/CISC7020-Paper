\begin{frame}{2.~Algorithm Overview}
\vspace*{-3mm}
\begin{columns}[T]
\begin{column}{0.44\textwidth}
\small
% The bullet list owns most of the left column, so its list margins are pulled
% in: \leftmargini shrinks the bullet indent (default 2.5em at 11pt) and a
% negative \rightmargin lets the text run a little past the column, into the
% 0.02\textwidth gutter between the columns. Net gain is about 2em per line.
\setlength{\leftmargini}{1.2em}
\setlength{\rightmargin}{-0.6em}
\begin{itemize}\setlength\itemsep{5pt}
\item Two programs: \texttt{tokbinR} at the root, \texttt{tokbinV} elsewhere
\item Root releases tokens; duplicates \textbf{merge}, never multiply
\item One \emph{bit} per vertex, flipped on every arrival
\item Pass on only if the neighbour's bit allows
\item A token has a bounded \textbf{lifespan}
\item \textbf{Cycles} can no longer trap a token
\item Settles in $O(2^{\varepsilon})$ rounds, never silent
\end{itemize}
\end{column}

\begin{frame}{4.~Kinship structure}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
\footnotesize
\setlength{\abovedisplayskip}{1pt}\setlength{\belowdisplayskip}{1pt}
\setlength{\abovedisplayshortskip}{1pt}\setlength{\belowdisplayshortskip}{1pt}
\tknum{1} The rank of $v$ turns its neighbourhood into a family:
\[
\begin{aligned}
\Pa(v)&=\{\,u\in\Neig(v)\mid \rk_u=(\rk_v-1)\bmod 3\,\}\\
\Ch(v)&=\{\,u\in\Neig(v)\mid \rk_u=(\rk_v+1)\bmod 3\,\}\\
\Si(v)&=\{\,u\in\Neig(v)\mid \rk_u=\rk_v\,\}
\end{aligned}
\]
Ranks live on a 3-cycle, so only the two ``adjacent'' levels are meaningful.

\vspace{1.5mm}
\tknum{2} \textbf{Reset state} --- an error makes $v$ disconnect itself:
\[
\mathit{Reset}(v)\;\equiv\;\rk_v=\bot\;\wedge\;t_v=\mathit{false}\;\wedge\;b_v=\bz
\]
A reset node has \emph{no} family: $\Pa(v)=\Ch(v)=\Si(v)=\emptyset$.

\vspace{1.5mm}
\tknum{3} \textbf{Parent selection} (no identifiers, so use ports):
\[
\mathrm{parent}(v)=\operatorname*{arg\,min}_{u\in\Pa(v)}\ \pi_v(u)
\]
computed \emph{on the fly}, never stored.

\[
\begin{array}{@{}l@{\quad}r@{\,}l@{}}
\tknum{1} & \Rer      &:\;b_v\!\neq\!\top\wedge Er(v)\\
          &           &\quad\phantom{:}\;\longrightarrow t_v\!:=\!w;\,b_v\!:=\!\top\\[1pt]
\tknum{2} & \Rreset   &:\;b_v\!=\!\top\vee Er_{var}(v)\\
          &           &\quad\phantom{:}\;\longrightarrow \rk_v\!:=\!\bot;\,t_v\!:=\!f;\,b_v\!:=\!\bz\\[1pt]

\tknum{4} & \Rjoin    &:\;\neg Er(v)\wedge\rk_v\!=\!\bot\wedge\mathit{TakeR}(v)\\
          &           &\quad\phantom{:}\;\longrightarrow \rk_v\!:=\!\rk_t(v);\,b_v\!:=\!\bo\\[1pt]
\tknum{5} & \Rtok     &:\;\neg Er(v)\wedge\rk_v\!\neq\!\bot\wedge t_v\!=\!f\wedge\mathit{TakeP}(v)\\
          &           &\quad\phantom{:}\;\longrightarrow t_v\!:=\!\mathit{true}\\[1pt]
\tknum{6} & \Radd     &:\;\neg Er(v)\wedge t_v\!=\!\mathit{true}\\
          &           &\quad\phantom{:}\;\longrightarrow t_v\!:=\!w;\,b_v\!:=\!(b_v\!+\!1)\bmod 2\\[1pt]
\tknum{7} & \Rready   &:\;\neg Er(v)\wedge t_v\!=\!w\\
          &           &\quad\phantom{:}\;\longrightarrow t_v\!:=\!f
\end{array}
\]
\end{column}

\begin{frame}{11.~Conclusion}
\vspace*{-2mm}
\small
\begin{itemize}\setlength\itemsep{5pt}
\item \textbf{A first.} Constant-space, deterministic, self-stabilizing BFS
      construction in semi-uniform \emph{synchronous} networks of arbitrary
      topology; no global parameter ($n$, $\Delta$, $\varepsilon$) is ever needed.
\item \textbf{A new technique.} Point-to-neighbour and distance-to-the-root are
      unavailable here, so an \emph{infinite token stream} breaks the cycles and
      rebuilds the BFS layers --- of independent interest for other tasks.
\item \textbf{A by-product.} Running \texttt{TokBin}, every node takes the colour
      opposite to its parent: an optimal $2$-colouring of bipartite graphs in
      $O(1)$ bits, against $\Omega(\log n)$ in earlier solutions.
\item \textbf{Open problem I: asynchrony.} The algorithm assumes lock-step
      synchrony; extending the token stream to an asynchronous setting, and any
      synchronizer that fits in constant space, remains unclear.
\item \textbf{Open problem II: time.} Space is constant but stabilization costs
      $O(2^{\varepsilon})$ rounds; enlarging the domain of $b$ saves tokens yet
      stays exponential. Constant space \emph{and} polynomial time: future work.
\end{itemize}
\end{frame}
