\begin{frame}{2.~Algorithm Overview: tokens with a limited lifespan}
\vspace*{-2mm}
\begin{columns}[T]
\begin{column}{0.45\textwidth}
\small
\begin{itemize}\setlength\itemsep{4pt}
\item $\mathtt{TokBinR}$ at $r$ $\parallel$ $\mathtt{TokBinV}$ at others
\item root floods tokens; tokens from several parents are \textbf{merged}
\item one bit per node, flipped at each reception: $\;b_v\!\leftarrow\!\bar b_v$
\item forward $\iff b_u=\bo$ $\Rightarrow$ bounded token lifespan
\item no cycle can sustain a token forever
\item stabilization time $O(2^{\varepsilon})$
\item $36$ states $=$ $6$ bits; the algorithm is \emph{not silent}
\end{itemize}
\end{column}

% 5 -- KINSHIP AND RESET

\begin{frame}{4.~Terms: kinship structure and the reset state}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
\footnotesize
\setlength{\abovedisplayskip}{1pt}\setlength{\belowdisplayskip}{1pt}
\setlength{\abovedisplayshortskip}{1pt}\setlength{\belowdisplayshortskip}{1pt}
The rank of $v$ turns its neighbourhood into a family:
\[
\begin{aligned}
\Pa(v)&=\{\,u\in\Neig(v)\mid \rk_u=(\rk_v-1)\bmod 3\,\}\\
\Ch(v)&=\{\,u\in\Neig(v)\mid \rk_u=(\rk_v+1)\bmod 3\,\}\\
\Si(v)&=\{\,u\in\Neig(v)\mid \rk_u=\rk_v\,\}
\end{aligned}
\]
A neighbour of rank $\rk_v+2$ is \emph{not} a parent --- ranks live on a
3-cycle, so only the two ``adjacent'' levels are meaningful.

\vspace{1.5mm}
\textbf{Reset state} --- an error makes $v$ disconnect itself:
\[
\mathit{Reset}(v)\;\equiv\;\rk_v=\bot\;\wedge\;t_v=\mathit{false}\;\wedge\;b_v=\bz
\]
A reset node has \emph{no} family: $\Pa(v)=\Ch(v)=\Si(v)=\emptyset$.

\vspace{1.5mm}
\textbf{Parent selection} (no identifiers, so use ports):
\[
\mathrm{parent}(v)=\operatorname*{arg\,min}_{u\in\Pa(v)}\ \pi_v(u)
\]
computed \emph{on the fly}, never stored.
\end{column}

\begin{frame}{11.~Conclusion: what we built, and what is left}
\vspace*{-2mm}
\small
\begin{itemize}\setlength\itemsep{5pt}
\item \textbf{A first.} Constant-space, deterministic, self-stabilizing BFS
      construction in semi-uniform \emph{synchronous} networks of arbitrary
      topology; no global parameter ($n$, $\Delta$, $\varepsilon$) is ever needed.
\item \textbf{A new technique.} Point-to-neighbour and distance-to-the-root are
      unavailable here, so an \emph{infinite token stream} breaks the cycles and
      rebuilds the BFS layers --- of independent interest for other tasks.
\item \textbf{A by-product.} Running \texttt{TokBin}, every node takes the colour
      opposite to its parent: an optimal $2$-colouring of bipartite graphs in
      $O(1)$ bits, against $\Omega(\log n)$ in earlier solutions.
\item \textbf{Open problem I: asynchrony.} The algorithm assumes lock-step
      synchrony; extending the token stream to an asynchronous setting, and any
      synchronizer that fits in constant space, remains unclear.
\item \textbf{Open problem II: time.} Space is constant but stabilization costs
      $O(2^{\varepsilon})$ rounds; enlarging the domain of $b$ saves tokens yet
      stays exponential. Constant space \emph{and} polynomial time: future work.
\end{itemize}
\end{frame}
